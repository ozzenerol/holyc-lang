Example projects

Word frequency

Reads a text file and prints its most common words. It shows reading a file, scanning characters, counting with a StrMap, ranking with PtrVecSort, and reading command-line arguments.

cd docs/examples/wordfreq
hcc main.HC -o wordfreq && ./wordfreq sample.txt
// Word frequency: reads a text file and prints its most common words.
//
//   hcc main.HC -o wordfreq && ./wordfreq sample.txt
//   hcc -jit main.HC sample.txt
//
// Shows: reading files, scanning strings, StrMap, PtrVec sorting,
// classes and command-line arguments.

#define TOP_N    8
#define MAX_WORD 64

class WordCount
{
  U8 *word;
  I64 count;
};

// Sort predicate: TRUE when a ranks above b. Most frequent first, then
// alphabetical so ties have a stable order.
I64 RanksHigher(U0 *_a, U0 *_b)
{
  WordCount *a = _a;
  WordCount *b = _b;
  if (a->count != b->count) {
    return a->count > b->count;
  }
  return StrCmp(a->word, b->word) < 0;
}

Bool IsLetter(U8 c)
{
  return ('a' <= c <= 'z') || ('A' <= c <= 'Z');
}

U0 CountWord(StrMap *counts, U8 *word)
{
  WordCount *wc = StrMapGet(counts, word);
  if (wc == NULL) {
    wc = MAlloc(sizeof(WordCount));
    wc->word = StrNew(word);       // the map keeps this pointer as its key
    wc->count = 0;
    StrMapSet(counts, wc->word, wc);
  }
  wc->count++;
}

I32 Main(I32 argc, U8 **argv)
{
  U8 *path = "sample.txt";
  if (argc > 1) {
    path = argv[1];
  }

  I64 size;
  U8 *text = FileRead(path, &size);
  if (text == NULL) {
    "cannot read %s\n", path;
    return 1;
  }

  StrMap *counts = StrMapNew();
  U8 word[MAX_WORD];
  I64 len = 0, total = 0;

  // One pass over the text; i == size flushes the final word.
  for (I64 i = 0; i <= size; i++) {
    U8 c = 0;
    if (i < size) c = text[i];
    if (IsLetter(c)) {
      if (len < MAX_WORD - 1) {
        word[len++] = ToLower(c);
      }
    } else if (len > 0) {
      word[len] = 0;
      CountWord(counts, word);
      total++;
      len = 0;
    }
  }

  PtrVec *ranked = PtrVecNew();
  I64 idx = 0;
  WordCount *wc;
  while (StrMapValueIter(counts, &idx, &wc)) {
    PtrVecPush(ranked, wc);
  }
  PtrVecSort(ranked, &RanksHigher);

  "%d words, %d distinct\n", total, ranked->size;
  for (I64 i = 0; i < TOP_N && i < ranked->size; i++) {
    WordCount *top = ranked->entries[i];
    "%2d. %-10s %d\n", i + 1, top->word, top->count;
  }

  Free(text);
  return 0;
}

Bank

Accounts with monthly interest and a transaction log. A savings account inherits from Account and swaps in its own interest function through a function pointer. Errors are exceptions, and a failed transfer rolls itself back. Money is kept in integer cents, so no rounding errors creep in.

cd docs/examples/bank
hcc main.HC -o bank && ./bank
// Bank: accounts, monthly interest and a transaction log.
//
//   hcc main.HC -o bank && ./bank
//   hcc -jit main.HC
//
// Shows: classes and inheritance, function pointers as "methods",
// exceptions for errors, linked lists, and integer money (cents).

#define ERR_FUNDS  'Funds'
#define ERR_FROZEN 'Frozen'

class Account
{
  U8 *owner;
  I64 balance;          // in cents
  Bool frozen;
  // Per-kind behaviour: interest earned in a month.
  I64 (*interest)(Account *a);
};

class Savings : Account
{
  I64 rate_bp;          // annual rate in basis points (1/100 of a percent)
};

class Txn
{
  U8 *owner;
  U8 *what;
  I64 amount;
};

List *ledger;

I64 NoInterest(Account *a)
{
  return 0;
}

I64 SavingsInterest(Account *a)
{
  Savings *s = a;
  return s->balance * s->rate_bp / 10000 / 12;
}

U0 InitAccount(Account *a, U8 *owner, I64 cents)
{
  a->owner = owner;
  a->balance = cents;
  a->frozen = FALSE;
  a->interest = &NoInterest;
}

U0 Record(Account *a, U8 *what, I64 amount)
{
  Txn *t = MAlloc(sizeof(Txn));
  t->owner = a->owner;
  t->what = what;
  t->amount = amount;
  ListAppend(ledger, t);
}

U0 Withdraw(Account *a, I64 cents)
{
  if (a->frozen) throw(ERR_FROZEN);
  if (cents > a->balance) throw(ERR_FUNDS);
  a->balance -= cents;
  Record(a, "withdraw", -cents);
}

U0 Deposit(Account *a, I64 cents)
{
  if (a->frozen) throw(ERR_FROZEN);
  a->balance += cents;
  Record(a, "deposit", cents);
}

// Moves money between accounts, or neither changes.
Bool Transfer(Account *from, Account *to, I64 cents)
{
  Bool withdrawn = FALSE;
  try {
    Withdraw(from, cents);
    withdrawn = TRUE;
    Deposit(to, cents);
  } catch {
    if (withdrawn) {                 // the deposit failed: put it back
      from->balance += cents;
      Record(from, "refund", cents);
    }
    U8 *why = "account frozen";
    if (Fs->except_ch == ERR_FUNDS) why = "insufficient funds";
    "  transfer %s -> %s refused: %s\n", from->owner, to->owner, why;
    return FALSE;
  }
  return TRUE;
}

U0 PrintMoney(I64 cents)
{
  if (cents < 0) {
    "-";
    cents = -cents;
  }
  "%d.%02d", cents / 100, cents % 100;
}

U0 Statement(Account *a)
{
  "  %-6s ", a->owner;
  PrintMoney(a->balance);
  if (a->frozen) " (frozen)";
  "\n";
}

U0 Main()
{
  ledger = ListNew();

  Account alice;
  InitAccount(&alice, "alice", 120000);

  Savings bob;
  InitAccount(&bob, "bob", 50000);
  bob.rate_bp = 450;                 // 4.5% a year
  bob.interest = &SavingsInterest;

  Account carol;
  InitAccount(&carol, "carol", 2500);

  Account *all[3];
  all[0] = &alice;
  all[1] = &bob;
  all[2] = &carol;

  "Opening balances\n";
  for (I64 i = 0; i < 3; i++) Statement(all[i]);

  "Transfers\n";
  Transfer(&alice, &bob, 30000);
  Transfer(&carol, &alice, 9999);    // carol only has 25.00
  carol.frozen = TRUE;
  Transfer(&bob, &carol, 100);

  "Month end\n";
  for (I64 i = 0; i < 3; i++) {
    Account *a = all[i];
    I64 earned = a->interest(a);
    if (earned > 0) {
      a->balance += earned;
      Record(a, "interest", earned);
    }
    Statement(a);
  }

  "Ledger (%d entries)\n", ListCount(ledger);
  for (List *it = ledger->next; it != ledger; it = it->next) {
    Txn *t = it->value;
    "  %-6s %-9s ", t->owner, t->what;
    PrintMoney(t->amount);
    "\n";
  }
}

Game of Life

Conway's Game of Life on a small board that wraps at the edges, with a glider and a blinker. It shows 2D arrays, nested loops, % for wrap-around and MemCpy on whole arrays.

cd docs/examples/life
hcc main.HC -o life && ./life
// Conway's Game of Life on a small wrapping board.
//
//   hcc main.HC -o life && ./life
//   hcc -jit main.HC
//
// Shows: 2D arrays, nested loops, the % operator for wrap-around,
// #define constants and building output a line at a time.

#define W     12
#define H     8
#define STEPS 4

U8 board[H][W];
U8 next[H][W];

U0 Place(U8 *pattern, I64 top, I64 left)
{
  // `pattern` rows are separated by '/'; '#' is a live cell.
  I64 r = top, c = left;
  for (U8 *p = pattern; *p; p++) {
    if (*p == '/') {
      r++;
      c = left;
    } else {
      board[r][c] = *p == '#';
      c++;
    }
  }
}

I64 Neighbours(I64 r, I64 c)
{
  I64 n = 0;
  for (I64 dr = -1; dr <= 1; dr++) {
    for (I64 dc = -1; dc <= 1; dc++) {
      if (dr == 0 && dc == 0) continue;
      n += board[(r + dr + H) % H][(c + dc + W) % W];
    }
  }
  return n;
}

U0 Step()
{
  for (I64 r = 0; r < H; r++) {
    for (I64 c = 0; c < W; c++) {
      I64 n = Neighbours(r, c);
      // Born with 3 neighbours, survives with 2 or 3.
      next[r][c] = n == 3 || (board[r][c] && n == 2);
    }
  }
  MemCpy(board, next, sizeof(board));
}

I64 Population()
{
  I64 alive = 0;
  for (I64 r = 0; r < H; r++) {
    for (I64 c = 0; c < W; c++) {
      alive += board[r][c];
    }
  }
  return alive;
}

U0 Show(I64 gen)
{
  "generation %d, %d alive\n", gen, Population;
  U8 line[W + 1];
  line[W] = 0;
  for (I64 r = 0; r < H; r++) {
    for (I64 c = 0; c < W; c++) {
      line[c] = '.';
      if (board[r][c]) line[c] = '#';
    }
    "%s\n", line;
  }
}

U0 Main()
{
  Place(".#./..#/###", 0, 0);   // a glider, top left
  Place("###", 5, 7);           // a blinker, bottom right
  for (I64 gen = 0; gen <= STEPS; gen++) {
    Show(gen);
    Step;
  }
}

Echo server

A multithreaded TCP echo server and its test clients in one program. The server binds to port 0 so the kernel picks a free port, reads it back with getsockname, and gives every connection its own thread. Four client threads connect at the same time and send their message in 5-byte pieces: TCP is a byte stream, so both sides loop until every byte is sent or received. A mutex guards the server's counters.

cd docs/examples/echo
hcc main.HC -o echo && ./echo
/* A multithreaded TCP echo server and the clients that test it, in one
 * program. The server listens on 127.0.0.1 on a port the kernel picks,
 * accepts each client on its own thread and sends back every byte it
 * receives. Four client threads connect at the same time, send their
 * message in small pieces, and check that it comes back unchanged. */

#define CLIENTS 4

class Server
{
  I32 fd;
  I64 port;
  pthread_mutex_t lock;   /* guards bytes and served */
  I64 bytes;
  I64 served;
};

class Conn
{
  Server *srv;
  I32 fd;
};

class Client
{
  I64 id;
  I64 port;
  U8 sent[64];
  U8 got[64];
  Bool ok;
};

/* The port the kernel gave a socket bound to port 0. */
I64 BoundPort(I32 fd)
{
  sockaddr_in addr;
  I64 len = sizeof(sockaddr_in);
  if (getsockname(fd, &addr(sockaddr *), &len) != 0) return -1;
  U16 p = addr.sin_port;              /* network byte order */
  return (p >> 8) | ((p & 0xFF) << 8);
}

/* send() may write less than asked: keep going until all of it is out. */
Bool SendAll(I32 fd, U8 *buf, I64 len)
{
  while (len > 0) {
    I64 n = send(fd, buf, len, 0);
    if (n <= 0) return FALSE;
    buf += n;
    len -= n;
  }
  return TRUE;
}

/* Read exactly len bytes, or fail if the peer closes first. */
Bool RecvAll(I32 fd, U8 *buf, I64 len)
{
  while (len > 0) {
    I64 n = recv(fd, buf, len, 0);
    if (n <= 0) return FALSE;
    buf += n;
    len -= n;
  }
  return TRUE;
}

U0 *ServeConn(Conn *c)
{
  U8 buf[256];
  I64 n, total = 0;
  while ((n = recv(c->fd, buf, sizeof(buf), 0)) > 0) {
    if (!SendAll(c->fd, buf, n)) break;
    total += n;
  }
  close(c->fd);

  pthread_mutex_lock(&c->srv->lock);
  c->srv->bytes += total;
  c->srv->served++;
  pthread_mutex_unlock(&c->srv->lock);
  Free(c);
  return NULL;
}

/* Accept CLIENTS connections, one handler thread each, then wait for them. */
U0 *AcceptLoop(Server *srv)
{
  pthread_t *handlers[CLIENTS];
  for (I64 i = 0; i < CLIENTS; i++) {
    Conn *c = MAlloc(sizeof(Conn));
    c->srv = srv;
    c->fd = accept(srv->fd, NULL, NULL);
    pthread_create(&handlers[i],,&ServeConn,c);
  }
  for (I64 i = 0; i < CLIENTS; i++) {
    pthread_join(handlers[i]);
  }
  return NULL;
}

U0 *RunClient(Client *cl)
{
  I32 fd = NetConnect("127.0.0.1", cl->port);
  if (fd <= 0) return NULL;

  snprintf(cl->sent, sizeof(cl->sent), "hello from client %d, over TCP", cl->id);
  I64 len = StrLen(cl->sent);

  /* Send in 5-byte pieces: TCP is a byte stream, the echo must still match. */
  for (I64 off = 0; off < len; off += 5) {
    I64 piece = len - off;
    if (piece > 5) piece = 5;
    SendAll(fd, cl->sent + off, piece);
  }
  cl->ok = RecvAll(fd, cl->got, len);
  cl->got[len] = '\0';
  close(fd);
  return NULL;
}

U0 Main()
{
  Server srv;
  MemSet(&srv, 0, sizeof(Server));
  pthread_mutex_init(&srv.lock);

  srv.fd = NetCreateServer(0, "127.0.0.1", CLIENTS);
  if (srv.fd <= 0) {
    "could not start the server\n";
    return;
  }
  srv.port = BoundPort(srv.fd);
  "server listening on 127.0.0.1 (port chosen by the kernel)\n";

  pthread_t *acceptor;
  pthread_create(&acceptor,,&AcceptLoop,&srv);

  Client clients[CLIENTS];
  pthread_t *threads[CLIENTS];
  for (I64 i = 0; i < CLIENTS; i++) {
    MemSet(&clients[i], 0, sizeof(Client));
    clients[i].id = i + 1;
    clients[i].port = srv.port;
    pthread_create(&threads[i],,&RunClient,&clients[i]);
  }
  for (I64 i = 0; i < CLIENTS; i++) {
    pthread_join(threads[i]);
  }
  pthread_join(acceptor);
  close(srv.fd);

  for (I64 i = 0; i < CLIENTS; i++) {
    Client *cl = &clients[i];
    U8 *verdict = "MISMATCH";
    if (cl->ok && !StrCmp(cl->sent, cl->got)) verdict = "echoed";
    "client %d: %s \"%s\"\n", cl->id, verdict, cl->got;
  }
  "server handled %d connections, echoed %d bytes\n", srv.served, srv.bytes;
}

HTTP server

A small HTTP/1.0 server with a thread per connection and three routes: a greeting, an /add route that reads two numbers from the query string, and 404/405 replies for everything else. The client side sends five requests at once, then parses each status line and body with strchr, strstr and Atoi.

cd docs/examples/http
hcc main.HC -o http && ./http
/* A tiny HTTP/1.0 server with a thread per connection, and a client that
 * sends it a few requests at the same time. The server has three routes:
 *   GET /             a greeting
 *   GET /add?a=&b=    adds two numbers from the query string
 *   anything else     404, or 405 for a method other than GET
 * Each response says how long its body is and closes the connection. */

#define REQUESTS 5

class Server
{
  I32 fd;
  I64 port;
};

class Request
{
  U8 method[8];
  U8 path[128];
  I64 status;
  U8 body[128];
};

/* The port the kernel gave a socket bound to port 0. */
I64 BoundPort(I32 fd)
{
  sockaddr_in addr;
  I64 len = sizeof(sockaddr_in);
  if (getsockname(fd, &addr(sockaddr *), &len) != 0) return -1;
  U16 p = addr.sin_port;              /* network byte order */
  return (p >> 8) | ((p & 0xFF) << 8);
}

Bool SendAll(I32 fd, U8 *buf, I64 len)
{
  while (len > 0) {
    I64 n = send(fd, buf, len, 0);
    if (n <= 0) return FALSE;
    buf += n;
    len -= n;
  }
  return TRUE;
}

/* Read until the peer closes (or the buffer is full). */
I64 RecvUntilClose(I32 fd, U8 *buf, I64 cap)
{
  I64 got = 0, n;
  while (got < cap - 1 && (n = recv(fd, buf + got, cap - 1 - got, 0)) > 0) {
    got += n;
  }
  buf[got] = '\0';
  return got;
}

/* Value of `name` in a query string like "a=2&b=40", or 0. */
I64 QueryInt(U8 *query, U8 *name)
{
  I64 nlen = StrLen(name);
  U8 *p = query;
  while (p && *p) {
    if (!StrNCmp(p, name, nlen) && p[nlen] == '=') return Atoi(p + nlen + 1);
    p = strchr(p, '&');
    if (p) p++;
  }
  return 0;
}

U8 *StatusText(I64 status)
{
  switch (status) {
    case 200: return "OK";
    case 404: return "Not Found";
    case 405: return "Method Not Allowed";
  }
  return "Error";
}

U0 Respond(I32 fd, I64 status, U8 *body)
{
  U8 *msg = StrPrint(NULL,
      "HTTP/1.0 %d %s\r\nContent-Type: text/plain\r\n"
      "Content-Length: %d\r\nConnection: close\r\n\r\n%s",
      status, StatusText(status), StrLen(body), body);
  SendAll(fd, msg, StrLen(msg));
  Free(msg);
}

U0 *ServeConn(I64 fd)
{
  U8 req[1024];
  I64 got = 0, n;

  /* Read the request head: everything up to the blank line. */
  while (got < sizeof(req) - 1) {
    n = recv(fd, req + got, sizeof(req) - 1 - got, 0);
    if (n <= 0) break;
    got += n;
    req[got] = '\0';
    if (strstr(req, "\r\n\r\n")) break;
  }
  req[got] = '\0';

  /* "GET /add?a=2&b=40 HTTP/1.0": split out the method and the target. */
  U8 *sp1 = strchr(req, ' ');
  U8 *sp2 = NULL;
  if (sp1) sp2 = strchr(sp1 + 1, ' ');
  if (!sp1 || !sp2) {
    Respond(fd, 400, "bad request\n");
  } else {
    *sp1 = '\0';
    *sp2 = '\0';
    U8 *method = req, *target = sp1 + 1;
    U8 *query = strchr(target, '?');
    if (query) *query++ = '\0';

    if (StrCmp(method, "GET")) {
      Respond(fd, 405, "only GET is supported\n");
    } else if (!StrCmp(target, "/")) {
      Respond(fd, 200, "Hello from HolyC\n");
    } else if (!StrCmp(target, "/add") && query) {
      I64 a = QueryInt(query, "a"), b = QueryInt(query, "b");
      U8 sum[64];
      snprintf(sum, sizeof(sum), "%d + %d = %d\n", a, b, a + b);
      Respond(fd, 200, sum);
    } else {
      Respond(fd, 404, "no such page\n");
    }
  }
  close(fd);
  return NULL;
}

U0 *AcceptLoop(Server *srv)
{
  pthread_t *handlers[REQUESTS];
  for (I64 i = 0; i < REQUESTS; i++) {
    I64 fd = accept(srv->fd, NULL, NULL);
    pthread_create(&handlers[i],,&ServeConn,fd(U0 *));
  }
  for (I64 i = 0; i < REQUESTS; i++) {
    pthread_join(handlers[i]);
  }
  return NULL;
}

class Job
{
  I64 port;
  Request *req;
};

/* Client side: send one request and parse the status code and body. */
U0 *Fetch(Job *job)
{
  Request *r = job->req;
  I32 fd = NetConnect("127.0.0.1", job->port);
  if (fd <= 0) return NULL;

  U8 *msg = StrPrint(NULL, "%s %s HTTP/1.0\r\nHost: localhost\r\n\r\n",
                     r->method, r->path);
  SendAll(fd, msg, StrLen(msg));
  Free(msg);

  U8 resp[1024];
  RecvUntilClose(fd, resp, sizeof(resp));
  close(fd);

  U8 *sp = strchr(resp, ' ');                /* "HTTP/1.0 200 OK" */
  if (sp) r->status = Atoi(sp + 1);
  U8 *body = strstr(resp, "\r\n\r\n");
  if (body) {
    StrCpy(r->body, body + 4);
    I64 len = StrLen(r->body);
    if (len && r->body[len - 1] == '\n') r->body[len - 1] = '\0';
  }
  return NULL;
}

U0 Main()
{
  Server srv;
  srv.fd = NetCreateServer(0, "127.0.0.1", REQUESTS);
  if (srv.fd <= 0) {
    "could not start the server\n";
    return;
  }
  srv.port = BoundPort(srv.fd);

  pthread_t *acceptor;
  pthread_create(&acceptor,,&AcceptLoop,&srv);

  Request reqs[REQUESTS];
  MemSet(reqs, 0, sizeof(reqs));
  StrCpy(reqs[0].method, "GET");  StrCpy(reqs[0].path, "/");
  StrCpy(reqs[1].method, "GET");  StrCpy(reqs[1].path, "/add?a=2&b=40");
  StrCpy(reqs[2].method, "GET");  StrCpy(reqs[2].path, "/add?b=-7&a=100");
  StrCpy(reqs[3].method, "GET");  StrCpy(reqs[3].path, "/missing");
  StrCpy(reqs[4].method, "POST"); StrCpy(reqs[4].path, "/");

  /* All five requests are in flight at once, each on its own thread. */
  Job jobs[REQUESTS];
  pthread_t *threads[REQUESTS];
  for (I64 i = 0; i < REQUESTS; i++) {
    jobs[i].port = srv.port;
    jobs[i].req = &reqs[i];
    pthread_create(&threads[i],,&Fetch,&jobs[i]);
  }
  for (I64 i = 0; i < REQUESTS; i++) {
    pthread_join(threads[i]);
  }
  pthread_join(acceptor);
  close(srv.fd);

  for (I64 i = 0; i < REQUESTS; i++) {
    Request *r = &reqs[i];
    "%-4s %-16s -> %d %-18s %s\n", r->method, r->path, r->status,
      StatusText(r->status), r->body;
  }
}

Key-value store

A server speaking a line protocol (SET, GET, INCR, DEL, COUNT), with every client on its own thread and all of them sharing one StrMap behind a mutex. Four clients run at once and each bumps the same counter 250 times; the final count of 1000 shows no update was lost. Remove the lock and the threads corrupt the map. It also shows splitting a TCP stream into lines and letting the map free its keys and values.

cd docs/examples/kvstore
hcc main.HC -o kvstore && ./kvstore
/* A key-value store server speaking a line protocol over TCP:
 *   SET <key> <value>   ->  OK
 *   GET <key>           ->  the value, or (nil)
 *   INCR <key>          ->  the new number (a missing key counts as 0)
 *   DEL <key>           ->  1 if it existed, else 0
 *   COUNT               ->  how many keys there are
 * Every client gets its own thread, and all of them share one StrMap
 * guarded by a mutex. Four clients run at once: each stores its name and
 * bumps a shared counter 250 times. Without the lock the threads would
 * corrupt the map (removing it makes this program crash), and increments
 * would be lost. A last client then reads everything back. */

#define WORKERS 4
#define BUMPS   250

class Store
{
  StrMap *map;
  pthread_mutex_t lock;
};

class Server
{
  I32 fd;
  I64 port;
  Store *store;
};

class Conn
{
  Store *store;
  I32 fd;
  U8 buf[512];   /* bytes read but not yet split into lines */
  I64 len;
};

/* The port the kernel gave a socket bound to port 0. */
I64 BoundPort(I32 fd)
{
  sockaddr_in addr;
  I64 len = sizeof(sockaddr_in);
  if (getsockname(fd, &addr(sockaddr *), &len) != 0) return -1;
  U16 p = addr.sin_port;              /* network byte order */
  return (p >> 8) | ((p & 0xFF) << 8);
}

Bool SendAll(I32 fd, U8 *buf, I64 len)
{
  while (len > 0) {
    I64 n = send(fd, buf, len, 0);
    if (n <= 0) return FALSE;
    buf += n;
    len -= n;
  }
  return TRUE;
}

/* Copy the next '\n'-terminated line into out (without the newline),
 * reading more from the socket as needed. FALSE when the peer is gone. */
Bool ReadLine(I32 fd, U8 *buf, I64 *_len, U8 *out, I64 cap)
{
  while (TRUE) {
    for (I64 i = 0; i < *_len; i++) {
      if (buf[i] == '\n') {
        I64 n = i;
        if (n >= cap) n = cap - 1;
        MemCpy(out, buf, n);
        out[n] = '\0';
        MemCpy(buf, buf + i + 1, *_len - i - 1);
        *_len -= i + 1;
        return TRUE;
      }
    }
    if (*_len >= 512) return FALSE;    /* line too long */
    I64 got = recv(fd, buf + *_len, 512 - *_len, 0);
    if (got <= 0) return FALSE;
    *_len += got;
  }
}

U0 FreeStr(U0 *p)
{
  Free(p);
}

/* Run one command against the store; the reply goes in out. */
U0 Execute(Store *st, U8 *line, U8 *out, I64 cap)
{
  U8 *cmd = line, *key = NULL, *val = NULL;
  key = strchr(cmd, ' ');
  if (key) {
    *key++ = '\0';
    val = strchr(key, ' ');
    if (val) *val++ = '\0';
  }

  pthread_mutex_lock(&st->lock);
  if (!StrCmp(cmd, "SET") && key && val) {
    StrMapDelete(st->map, key);                  /* frees an old key/value */
    StrMapSet(st->map, StrNew(key), StrNew(val));
    snprintf(out, cap, "OK");
  } else if (!StrCmp(cmd, "GET") && key) {
    U8 *v = StrMapGet(st->map, key);
    if (v) snprintf(out, cap, "%s", v);
    else   snprintf(out, cap, "(nil)");
  } else if (!StrCmp(cmd, "INCR") && key) {
    I64 n = 0;
    U8 *v = StrMapGet(st->map, key);
    if (v) n = Atoi(v);
    n++;
    StrMapDelete(st->map, key);
    StrMapSet(st->map, StrNew(key), StrPrint(NULL, "%d", n));
    snprintf(out, cap, "%d", n);
  } else if (!StrCmp(cmd, "DEL") && key) {
    snprintf(out, cap, "%d", StrMapDelete(st->map, key));
  } else if (!StrCmp(cmd, "COUNT")) {
    snprintf(out, cap, "%d", st->map->size);
  } else {
    snprintf(out, cap, "ERR unknown command");
  }
  pthread_mutex_unlock(&st->lock);
}

U0 *ServeConn(Conn *c)
{
  U8 line[256], reply[300];
  while (ReadLine(c->fd, c->buf, &c->len, line, sizeof(line))) {
    Execute(c->store, line, reply, sizeof(reply) - 1);
    I64 n = StrLen(reply);
    reply[n] = '\n';
    if (!SendAll(c->fd, reply, n + 1)) break;
  }
  close(c->fd);
  Free(c);
  return NULL;
}

/* Serve exactly `clients` connections, each on its own thread. */
U0 *AcceptLoop(Server *srv)
{
  pthread_t *handlers[WORKERS + 1];
  for (I64 i = 0; i < WORKERS + 1; i++) {
    Conn *c = CAlloc(sizeof(Conn));
    c->store = srv->store;
    c->fd = accept(srv->fd, NULL, NULL);
    pthread_create(&handlers[i],,&ServeConn,c);
  }
  for (I64 i = 0; i < WORKERS + 1; i++) {
    pthread_join(handlers[i]);
  }
  return NULL;
}

/* ---- client side ---- */

class Session
{
  I32 fd;
  U8 buf[512];
  I64 len;
};

Bool SessionOpen(Session *s, I64 port)
{
  MemSet(s, 0, sizeof(Session));
  s->fd = NetConnect("127.0.0.1", port);
  return s->fd > 0;
}

/* Send one command and wait for its one-line reply. */
U0 Call(Session *s, U8 *cmd, U8 *reply, I64 cap)
{
  SendAll(s->fd, cmd, StrLen(cmd));
  SendAll(s->fd, "\n", 1);
  if (!ReadLine(s->fd, s->buf, &s->len, reply, cap)) StrCpy(reply, "(closed)");
}

class Worker
{
  I64 id;
  I64 port;
};

U0 *RunWorker(Worker *w)
{
  Session s;
  U8 cmd[64], reply[64];
  if (!SessionOpen(&s, w->port)) return NULL;
  snprintf(cmd, sizeof(cmd), "SET worker:%d done-by-thread-%d", w->id, w->id);
  Call(&s, cmd, reply, sizeof(reply));
  for (I64 i = 0; i < BUMPS; i++) {
    Call(&s, "INCR hits", reply, sizeof(reply));
  }
  close(s.fd);
  return NULL;
}

U0 Show(Session *s, U8 *cmd)
{
  U8 reply[128];
  Call(s, cmd, reply, sizeof(reply));
  "%-16s -> %s\n", cmd, reply;
}

U0 Main()
{
  Store store;
  store.map = StrMapNew();
  StrMapSetFreeKey(store.map, &FreeStr);
  StrMapSetFreeValue(store.map, &FreeStr);
  pthread_mutex_init(&store.lock);

  Server srv;
  srv.store = &store;
  srv.fd = NetCreateServer(0, "127.0.0.1", WORKERS + 1);
  if (srv.fd <= 0) {
    "could not start the server\n";
    return;
  }
  srv.port = BoundPort(srv.fd);

  pthread_t *acceptor;
  pthread_create(&acceptor,,&AcceptLoop,&srv);

  Worker workers[WORKERS];
  pthread_t *threads[WORKERS];
  for (I64 i = 0; i < WORKERS; i++) {
    workers[i].id = i + 1;
    workers[i].port = srv.port;
    pthread_create(&threads[i],,&RunWorker,&workers[i]);
  }
  for (I64 i = 0; i < WORKERS; i++) {
    pthread_join(threads[i]);
  }
  "%d clients each sent INCR hits %d times\n", WORKERS, BUMPS;

  Session s;
  if (!SessionOpen(&s, srv.port)) {
    "could not connect\n";
    return;
  }
  Show(&s, "GET hits");
  Show(&s, "GET worker:1");
  Show(&s, "GET worker:4");
  Show(&s, "COUNT");
  Show(&s, "DEL worker:2");
  Show(&s, "DEL worker:2");
  Show(&s, "GET worker:2");
  Show(&s, "COUNT");
  Show(&s, "FLY away");
  close(s.fd);

  pthread_join(acceptor);
  close(srv.fd);
  StrMapRelease(store.map);
}
